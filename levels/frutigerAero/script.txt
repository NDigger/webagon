
// Pattern spawn conditions, uses level.onStep
const addPattern = async pKey => {
    const wallSpeed = level.getWallSpeedMult() * level.getDifficultyMult();
    const d = Math.max(wallSpeed, 7) * (45 - wallSpeed / 1.5);
    const de = d * 1.8;
    if (pKey === 0) await patterns.pRandomBarrage(Utils.mathRandom(4, 5), d * 1.1, de);
    else if (pKey === 1) await patterns.pAltBarrage(Utils.mathRandom(3, 6), d, de);
    else if (pKey === 2) await patterns.pTunnel(Utils.mathRandom(3, 5), d * 2.7, de, 1);
    else if (pKey === 3) await patterns.pLRBarrage(4, d, de);
    else if (pKey === 4) await patterns.pLeftRight(Utils.mathRandom(4, 5), d, de);
    else if (pKey === 5) await patterns.pSpiralBarrage(Utils.mathRandom(3, 6), d, de);
    else if (pKey === 6) await patterns.pDoubleSpiral(Utils.mathRandom(5, 7), d * 0.8, de);
    else if (pKey === 7) await patterns.pAltSpam(Utils.mathRandom(3, 4), 60, de);
    else if (pKey === 8) await patterns.pWallExSpam(Utils.mathRandom(3, 4), 60, de);
    else if (pKey === 9) await patterns.pTunnel(Utils.mathRandom(3, 5), d * 2, de, 3);
    else if (pKey === 10) await patterns.pWallExVortex(Utils.mathRandom(3, 4), d, de);
}

const pKeys = [0, 0, 0, 1, 1, 1, 2, 2, 3, 3, 4, 4, 5, 6, 7, 8, 9, 10];
let activeKeys = [];

// onInit is called on the first frame when level is created.
level.onInit = () => {
    level.setWallSpawnDistance(2000);
    level.setMainColor(new Color(255, 255, 255));
    level.setRadius(70);
    level.setRotationSpeed(0.15);
    level.setWallSpeedMult(5);
    level.setSides(7);
    level.set3dLayersCount(4);
    level.setIncrementTime(12);
    level.setWallSpeedIncrement(0.3);
    level.setRotationSpeedIncrement(0.015);
    level.set3dColor(new Color(255, 255, 255, 35));
    level.set3dFalloffColor(new Color(255, 255, 255, 0));
    level.setBackgroundSwapTime(.25);
    level.setScale(new Vector2(1, 1))
    level.setBackgroundDarkenUnevenChunkEnabled(false);
    level.set3dDistance(20);
    
    level.set3dDepthMult(0.5);
    level.set3dFalloffScale(new Vector2(.8, 2.1))
    level.setWallSkewLeft(40);
    level.setSkew(.5);
    update();
}

// onStep must be async and use delays in order to work. No delays may cause crash.
// Insert your wall spawn logic here
level.onStep = async () => {
    if (activeKeys.length === 0) activeKeys = pKeys.slice();
    const rndIndex = Math.floor(Math.random() * activeKeys.length)
    await level.distanceDelay(0);
    await addPattern(activeKeys.splice(rndIndex, 1)[0])
}

const update = () => {
    const p = Utils.pingPong(level.getTime() * 2.5) * 20;
    level.setBackgroundTileColors([
        new Color(105, 105 - p, 235 + p),
        new Color(90, 100 - p, 225 + p),
        new Color(80, 95 - p, 215 + p),
        new Color(70, 70 - p, 205 + p),
        new Color(80, 85 - p, 215 + p),
        new Color(90, 105 - p, 225 + p),
        new Color(105, 125 - p, 235 + p),
    ])

    const t = level.getTime();
    const sync = 1.-Utils.fract(level.getTime() * 2.5);

    level.setRadius(70 + sync*10);

    const wallScale = Math.sin(t * 20) * .04 + 1
    level.setWallScale(new Vector2(wallScale * (1 - sync * .05), wallScale * (1 - sync * .05)))
}
// onUpdate is called every frame.
level.onUpdate = ft => {
    update()
}

// onRender is called every frame. It works when player is died.
level.onRender = ft => {
    level.setSkew(Utils.pingPong(level.getTime()) * .5 + .5)
}

// onPreIncrement is called immediately when increment time is achieved
level.onPreIncrement = () => {}

// onIncrement is called every time walls are gone and level speed incremented
level.onIncrement = () => {}

// onDeath is called when main player of level object touches deadly wall side
level.onDeath = () => {}
