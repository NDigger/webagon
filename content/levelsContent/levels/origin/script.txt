// '../../patterns';

const addPattern = async pKey => {
    const diff = this.getDifficultyMult();
    const speed = this.getWallSpeedMult();
    const wallSpeed = speed * diff;
    const d = 500 * Math.max(1, wallSpeed / 8);
    const spiralD = d * speed * diff * .07;
    
    // if (pKey === 0) await pInverseBarrage(mathRandom(2, 3), d, d * 1.4);
    // else if (pKey === 1) await pSpiralBarrage(mathRandom(4, 6), d * .5, d);
    // else if (pKey === 2) await pSpiral(mathRandom(7, 9), spiralD, d * .8, 1);
    // else if (pKey === 3) await pTunnel(mathRandom(2, 3), d * 1.4, d * 1.2);
}

const pKeys = [0, 1, 2, 3];
let activeKeys = [];

const enableSwapOnHighSpeed = () => {
    const wasSwapEnabled = this.getSwapEnabled();
    this.getWallSpeedMult() * this.getDifficultyMult() > 5.8 && this.setSwapEnabled(true);
    if (wasSwapEnabled !== this.getSwapEnabled())
        this.showMessage('Speedmult > 5.8\nSwap enabled!', 2)
}

this.onInit = () => {
    this.setMainColor(new Color(255, 0, 0));
    this.setRadius(70);
    this.setRotationSpeed(0.035);
    this.setWallSpeedMult(2.4);
    this.setSides(this.getDifficultyMult() > 2 ? 6 : 5);
    this.set3dLayersCount(8);
    this.set3dDistance(5);
    this.setWallSpeedIncrement(0.2);
    this.setRotationSpeedIncrement(0.015);
    this.setSkew(0);
    this.setPlayerSwapReloadTime(.3);
    enableSwapOnHighSpeed();
}

// onStep must be async and use delays in order to work. No delays may cause crash.
this.onStep = async () => {
    if (activeKeys.length === 0) activeKeys = pKeys.slice();
    const rndIndex = Math.floor(Math.random() * activeKeys.length)
    await addPattern(activeKeys.splice(rndIndex, 1)[0])
    await this.distanceDelay(0);
}

// onUpdate is called every frame.
this.onUpdate = ft => {
    this.setBackgroundTileColors([
        new Color(75, 75, 75),
        new Color(85, 85, 85),
    ])
    const t = this.getTime();
    this.setMainColor(Color.hsvToRgb(t * 0.2, 1, 1))
    this.setScale(new Vector2(
        Utils.pingPong(t * .5) * .2 + 1, 
        Utils.pingPong(t * .5) * .2 + 1)
    )
}

this.onIncrement = () => {
    enableSwapOnHighSpeed();
    if (this.getDifficultyMult() > 2) return
    this.setSides(mathRandom(5, 6))
}

this.onRender = ft => {
    this.setSkew(Utils.pingPong(this.getTime()*.07)*.5)
}
