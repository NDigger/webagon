const getRandomSide = () => Math.floor(Math.random() * this.getSides()); 
const getRandomDir = () => Math.random() < .5 ? -1 : 1;
const getShift = () => Math.floor(Math.random() * 2);

const extraTunnelThickness = 40;
const barrage = side => {
    for(let i = 0; i < this.getSides() - 1; i++) {
        this.createWall(i + side, 40)
    }
}

const alt = async side => {
    for(let i = 0; i < this.getSides(); i+=2) {
        this.createWall(i + side, 40);
    }
}

const wallEx = async side => {
    if (this.getSides() % 2 === 0) {
        for(let i = 0; i < this.getSides()/2-1; i++) {
            this.createWall(i + side, 40);
            this.createWall(i + side + this.getSides()/2, 40);
        }
    } else {
        for(let i = 0; i < this.getSides()/2-1; i++) this.createWall(i + side, 40)
        for(let i = this.getSides()/2+1; i < this.getSides()-1; i++) this.createWall(i + side, 40)
    }
}

const pInverseBarrage = async (times, delay, delayEnd = 0) => {
    const side = getRandomSide();
    for(let i = 0; i < times; i++) {
        barrage((i%2)*this.getSides()/2+side);
        if (i !== times - 1) await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}

const pSpiral = async (times, delay, delayEnd = 0, extra = 1) => {
    const side = getRandomSide();
    const dir = getRandomDir();
    for(let i = 0; i < times; i++) {
        for(let k = 0; k < extra; k++) this.createWall((i+k) * dir + side, delay)
        await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}

const pDoubleSpiral = async (times, delay, delayEnd = 0, extra = 1) => {
    const side = getRandomSide();
    const dir = getRandomDir();
    for(let i = 0; i < times; i++) {
        for(let k = 0; k < extra; k++) {
            this.createWall((i+k) * dir + side, delay);
            this.createWall((i+k+this.getSides()/2) * dir + side, delay);
        }
        await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}

const pDoubleInverseSpiral = async (times, delay, delayEnd = 0, extra = 1) => {
    let side = getRandomSide();
    const dir = getRandomDir();
    for(let i = 0; i < times; i++) {
        side += dir;
        for(let k = 0; k < extra; k++) {
            this.createWall(side+k, delay);
            this.createWall(side+k+this.getSides()/2, delay);
        }
        await this.distanceDelay(delay);
    }
    for(let i = 0; i < times; i++) {
        side -= dir;
        for(let k = 0; k < extra; k++) {
            this.createWall(side+k, delay);
            this.createWall(side+k+this.getSides()/2, delay);
        }
        await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}


const pSpiralBarrage = async (times, delay, delayEnd = 0, step = 1) => {

    for(let i = 0; i < times; i++) {
        barrage(i * step * dir + side);
        if (i !== times - 1) await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}

const pLeftRight = async (times, delay, delayEnd = 0) => {
    const side = getRandomSide();
    for(let i = 0; i < times; i++) {
        if (i % 2 === 0) {
            barrage(side);
        } else {
            this.createWall(side - 1, 40)
        }
        if (i !== times - 1) await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}

const pLRBarrage = async (times, delay, delayEnd = 0, step = 1) => {
    const side = getRandomSide();
    const shift = getShift();
    for (let i = 0; i < times; i++) {
        barrage(side + ((i+shift) % 2 === 0 ? step : 0));
        if (i !== times - 1) await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
} 

const pWallExVortex = async (times, delay, delayEnd = 0) => {
    const side = getRandomSide();
    const dir = getRandomDir();
    let shift = 0;
    for (let i = 0; i < times; i++) {
        shift += dir;
        wallEx(side + shift);
        await this.distanceDelay(delay);
    }
    for (let i = 0; i < times - 1; i++) {
        shift -= dir;
        wallEx(side + shift);
        if (i !== times - 1) await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}

const pTunnel = async (times, delay, delayEnd = 0) => {
    const side = getRandomSide();
    const shift = getShift();
    for(let i = 0; i < times; i++) {
        barrage(((i + shift) % 2 === 0) ? side : side - 2);
        if (i !== times - 1) {
            this.createWall(side - 2, delay + extraTunnelThickness);
            await this.distanceDelay(delay);
        }
    }
    await this.distanceDelay(delayEnd);
}

const pAltTunnel = async (times, delay, delayEnd = 0, extra = 1) => {
    const side = getRandomSide();
    const shift = getShift();
    for(let i = 0; i < times; i++) {
        alt((i % 2 === 0) ? side + shift : side - 1 + shift);
        for (let k = 0; k < extra; k++) this.createWall(side - 2 + k, i !== times - 1 ? delay + extraTunnelThickness : 40);
        if (i !== times - 1) await this.distanceDelay(delay);
    }
    await this.distanceDelay(delayEnd);
}

const pWallExTunnel = async (times, delay, delayEnd) => {
    const side = getRandomSide();
    const shift = getShift();
    for (let i = 0; i < times; i++) {
        wallEx((i + shift) % 2 === 0 ? side : side + 2);
        this.createWall(side, i !== times -1 ? delay + extraTunnelThickness : 40);
        this.createWall(side + this.getSides()/2, i !== times -1 ? delay + extraTunnelThickness : 40);
        if (i !== times - 1) await this.distanceDelay(delay)
    }
    await this.distanceDelay(delayEnd);
}

const pRandomLRBarrage = async (times, delay, delayEnd) => {
    let side = getRandomSide();
    for (let i = 0; i < times; i++) {
        side += getRandomDir();
        barrage(side);
        if (i !== times - 1) await this.distanceDelay(delay)
    }
    await this.distanceDelay(delayEnd)
}

const pRandomBarrage = async (times, delay, delayEnd) => {
    let side = getRandomSide();
    for (let i = 0; i < times; i++) {
        barrage(side);
        const inc = Math.floor(Math.random() * (this.getSides()/2))*getRandomDir();
        side += inc;
        if (i !== times - 1) await this.distanceDelay(Math.abs(inc) * delay / this.getSides() * 4 + 60)
    }
    await this.distanceDelay(delayEnd);
}
