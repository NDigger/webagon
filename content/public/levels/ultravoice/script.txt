const half = side => {
    for (let i = 0; i < level.getSides()/2; i++) level.createWall(i + side)
}

const specialLR = async (times, delay, delayEnd) => {
    let side = Utils.getRandomSide(level);
    for (let i = 0; i < times; i++) {
        half(side += level.getSides()/2);
        if (i !== times - 1) await level.distanceDelay(delay);
    }
    await level.distanceDelay(delayEnd);
}

const barrage = (side, thickness = 40) => {
    for(let i = 0; i < level.getSides() - 1; i++) {
        level.createWall(i + side, thickness)
    }
}


const pTunnel = async (times, delay, delayEnd = 0) => {
        const side = Utils.getRandomSide(level);
        const shift = Utils.getShift();
        const thickness = level.getWallSpeedMult() * level.getDifficultyMult() * 20;
        for(let i = 0; i < times; i++) {
            barrage(((i + shift) % 2 === 0) ? side : side - 2, thickness);
            if (i !== times - 1) {
                level.createWall(side - 2, delay + 40 + thickness);
                await level.distanceDelay(delay + thickness);
            }
        }
        await level.distanceDelay(delayEnd);
    }

// Pattern spawn conditions, uses level.onStep
const addPattern = async pKey => {
    const speedMult = level.getDifficultyMult() * level.getWallSpeedMult();
    const d = 230 * Math.max(speedMult/8.25, 1);
    const de = d * 2.3;
    if (pKey === 0) await patterns.pInverseBarrage(Utils.mathRandom(3, 5), de, de);
    else if (pKey === 1) await patterns.pSpiralBarrage(Utils.mathRandom(5, 9), d, de);
    else if (pKey === 2) await patterns.pSpiral(Utils.mathRandom(7, 9), d, d * 1.5, Utils.mathRandom(1, 2));
    else if (pKey === 3) await patterns.pRandomBarrage(Utils.mathRandom(4, 9), d, de);
    else if (pKey === 4) await pTunnel(Utils.mathRandom(4, 5), de, de);
    else if (pKey === 5) await patterns.pAltBarrage(Utils.mathRandom(4, 7), d * 1.5, de);
    else if (pKey === 6) await specialLR(Utils.mathRandom(4, 6), d * 1.5, de);
    else if (pKey === 7) await patterns.pLRBarrage(Utils.mathRandom(4, 6), d * 1.5, de);
    else if (pKey === 8) await patterns.pLeftRight(Utils.mathRandom(4, 6), d * 1.5, de);
    else if (pKey === 9) await patterns.pAltSpam(3, 80, de);
}

const pKeys = [0, 0, 1, 1, 1, 2, 2, 4, 5, 5, 5, 6, 6, 7, 7, 8, 8, 9];
let activeKeys = [];

// onInit is called on the first frame when level is created.
level.onInit = () => {
    level.setMainColor(new Color(255, 0, 0));
    level.setRadius(60);
    level.setRotationSpeed(0.3);
    level.setWallSpeedMult(5.5);
    level.setSides(4);
    level.set3dLayersCount(4);
    level.set3dDistance(100);
    level.setWallSpeedIncrement(0.2);
    level.setRotationSpeedIncrement(0.017);
    level.set3dDepthMult(0);
    level.setScale(new Vector2(1.4, 1.4));
    // level.setWallSkewRight(20);
    level.set3dFalloffColor(new Color(0, 0, 0, 0));
    // level.set3dFalloffScale(new Vector2(.5, .5))
    level.setIncrementTime(12);
    level.setIncrementSpinPower(.3);
}

// onStep must be async and use delays in order to work. No delays may cause crash.
// Insert your wall spawn logic here
level.onStep = async () => {
    if (activeKeys.length === 0) activeKeys = pKeys.slice();
    const rndIndex = Math.floor(Math.random() * activeKeys.length)
    await level.distanceDelay(0) // To prevent crush if no active keys avaliable
    await addPattern(activeKeys.splice(rndIndex, 1)[0])
}

// onUpdate is called every frame.
let bgSwapTime = 0;
level.onUpdate = ft => {
    bgSwapTime += ft;
    if (bgSwapTime > 1.7) {
        bgSwapTime = 0;
        level.setRotationSpeed(-level.getRotationSpeed());
    }
    level.setBackgroundRotationOffset(level.getRotation() * -2)
    const t = level.getTime() * 10;
    level.setBackgroundTileColors([
        Utils.interpolate(new Color(25, 25, 0), new Color(0, 0, 0), Utils.pingPong(t)),
        Color.BLACK()
    ])

    // level.setRadius(70 - Utils.fract(level.getTime() * 2.2) * 10);

    const mainColor = Utils.interpolate(new Color(255, 255, 0), new Color(255, 255, 255), Utils.pingPong(t));
    level.setMainColor(mainColor)
    level.set3dColor(new Color(mainColor.r, mainColor.g, mainColor.b, 105));

    const s = 1 - Utils.pingPong(level.getTime() * 2.2) * .2;
    level.setWallScale(new Vector2(s, s));
}

// onRender is called every frame. It works when player is died.
level.onRender = ft => {
    level.setSkew(Utils.pingPong(level.getTime() * 3) * .1 + .3)
}

// onPreIncrement is called immediately when increment time is achieved
level.onPreIncrement = () => {}

// onIncrement is called every time walls are gone and level speed incremented
level.onIncrement = () => {}

// onDeath is called when main player of level object touches deadly wall side
level.onDeath = () => {}
